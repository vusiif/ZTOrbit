cmake_minimum_required(VERSION 3.21.1)
project(ztgui_voice_client LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ZTGUIPluginApi 0.5 EXACT CONFIG REQUIRED)
find_package(ZTGUIMediaApi 2 CONFIG REQUIRED)
find_package(Qt6 6.8 REQUIRED COMPONENTS Network)
configure_file(plugin.json.in "${CMAKE_CURRENT_BINARY_DIR}/plugin.json" @ONLY)
qt_add_plugin(ztgui_voice_chat SHARED CLASS_NAME vusiif::ztgui::voice_chat::VoiceChatPlugin
    Plugin.cpp VoiceController.cpp VoiceController.h VoiceSession.cpp VoiceSession.h AudioDeviceCache.h VoiceConfig.cpp VoiceConfig.h VoiceSoundPlayer.cpp VoiceSoundPlayer.h)
target_include_directories(ztgui_voice_chat PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
target_link_libraries(ztgui_voice_chat PRIVATE ZTGUI::Api ZTGUI::MediaApi Qt6::Network)
qt_add_resources(ztgui_voice_chat voice_page PREFIX /ztgui/voice-chat FILES VoicePage.qml)
set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/sounds/join.wav" PROPERTIES QT_RESOURCE_ALIAS "sounds/join.wav")
set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/sounds/leave.wav" PROPERTIES QT_RESOURCE_ALIAS "sounds/leave.wav")
qt_add_resources(ztgui_voice_chat voice_sounds PREFIX /ztgui/voice-chat
    FILES sounds/join.wav sounds/leave.wav)
set_target_properties(ztgui_voice_chat PROPERTIES PREFIX "")
if(MSVC)
    target_compile_options(ztgui_voice_chat PRIVATE /utf-8)
endif()
option(VOICE_BUILD_TESTS "Build voice UI/transport tests" OFF)
if(VOICE_BUILD_TESTS)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test QuickControls2)
    qt_add_executable(voice_ui_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/voice-chat/VoiceUiTests.cpp"
        VoiceController.cpp VoiceController.h VoiceSession.cpp VoiceSession.h VoiceConfig.cpp VoiceConfig.h VoiceSoundPlayer.cpp VoiceSoundPlayer.h)
    target_link_libraries(voice_ui_tests PRIVATE ZTGUI::Api ZTGUI::MediaApi Qt6::Test Qt6::Network Qt6::QuickControls2)
    target_include_directories(voice_ui_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
    target_compile_definitions(voice_ui_tests PRIVATE VOICE_PAGE="${CMAKE_CURRENT_SOURCE_DIR}/VoicePage.qml" VOICE_TESTING)
    if(MSVC)
        target_compile_options(voice_ui_tests PRIVATE /utf-8)
    endif()
    add_test(NAME voice_ui COMMAND voice_ui_tests -platform offscreen)
endif()
ztgui_package_plugin(ztgui_voice_chat HOST client ID voice-chat NAME "语音聊天" VERSION "0.3.0"
    API "${ZTGUI_API_VERSION}" IID "vusiif.ztgui.api.PluginInterface/3.0"
    SERVICE_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/service-manifest.json")
install(FILES "$<TARGET_FILE_DIR:ztgui_voice_chat>/ztgui_voice_chat.ztplugin" DESTINATION plugins)

cmake_minimum_required(VERSION 3.21.1)
project(ztgui_voice_client LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ZTGUIPluginApi 0.4 EXACT CONFIG REQUIRED)
find_package(Qt6 6.8 REQUIRED COMPONENTS Network Multimedia)
get_target_property(qt_type Qt6::Core TYPE)
if(qt_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Voice plugin requires shared Qt")
endif()
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF)
FetchContent_Declare(opus
    URL https://downloads.xiph.org/releases/opus/opus-1.6.1.tar.gz
    URL_HASH SHA256=6ffcb593207be92584df15b32466ed64bbec99109f007c82205f0194572411a1)
FetchContent_MakeAvailable(opus)
configure_file(plugin.json.in "${CMAKE_CURRENT_BINARY_DIR}/plugin.json" @ONLY)
qt_add_plugin(ztgui_voice_chat SHARED CLASS_NAME vusiif::ztgui::voice_chat::VoiceChatPlugin
    Plugin.cpp VoiceController.cpp VoiceController.h VoiceSession.cpp VoiceSession.h
    AudioEngine.cpp AudioEngine.h AudioMath.h AudioDeviceCache.h)
target_include_directories(ztgui_voice_chat PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
target_link_libraries(ztgui_voice_chat PRIVATE ZTGUI::Api Qt6::Network Qt6::Multimedia Opus::opus)
qt_add_resources(ztgui_voice_chat voice_page PREFIX /ztgui/voice-chat FILES VoicePage.qml)
set_target_properties(ztgui_voice_chat PROPERTIES PREFIX "")
if(MSVC)
    target_compile_options(ztgui_voice_chat PRIVATE /utf-8)
endif()
option(VOICE_BUILD_TESTS "Build audio math and codec tests" OFF)
if(VOICE_BUILD_TESTS)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(voice_audio_tests "../../../tests/voice-chat/AudioTests.cpp")
    target_link_libraries(voice_audio_tests PRIVATE Qt6::Test Qt6::Multimedia Opus::opus)
    target_include_directories(voice_audio_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
    add_test(NAME voice_audio COMMAND voice_audio_tests)
    find_package(Qt6 REQUIRED COMPONENTS QuickControls2)
    qt_add_executable(voice_ui_tests "../../../tests/voice-chat/VoiceUiTests.cpp"
        VoiceController.cpp VoiceController.h VoiceSession.cpp VoiceSession.h AudioEngine.cpp AudioEngine.h)
    target_link_libraries(voice_ui_tests PRIVATE ZTGUI::Api Qt6::Test Qt6::Network Qt6::Multimedia Qt6::QuickControls2 Opus::opus)
    target_include_directories(voice_ui_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
    target_compile_definitions(voice_ui_tests PRIVATE VOICE_PAGE="${CMAKE_CURRENT_SOURCE_DIR}/VoicePage.qml")
    if(MSVC)
        target_compile_options(voice_ui_tests PRIVATE /utf-8)
    endif()
    add_test(NAME voice_ui COMMAND voice_ui_tests -platform offscreen)
endif()

ztgui_package_plugin(ztgui_voice_chat HOST client ID voice-chat NAME "语音聊天" VERSION "0.1.0" API "${ZTGUI_API_VERSION}" IID "vusiif.ztgui.api.PluginInterface/3.0"
    RUNTIME_QT_MODULES Multimedia LICENSE_FILES "licenses/Opus-COPYING.txt=${opus_SOURCE_DIR}/COPYING")
install(FILES "$<TARGET_FILE_DIR:ztgui_voice_chat>/ztgui_voice_chat.ztplugin" DESTINATION plugins)

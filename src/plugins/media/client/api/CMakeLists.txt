cmake_minimum_required(VERSION 3.21.1)
project(ZTOrbitMediaApi VERSION 2.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(ZTOrbitPluginApi 0.5 EXACT CONFIG REQUIRED COMPONENTS Services)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)
add_library(zto_media_types INTERFACE)
add_library(ZTO::MediaTypes ALIAS zto_media_types)
set_target_properties(zto_media_types PROPERTIES EXPORT_NAME MediaTypes)
target_link_libraries(zto_media_types INTERFACE Qt6::Core)
target_compile_features(zto_media_types INTERFACE cxx_std_17)
target_include_directories(zto_media_types INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
add_library(zto_media_api INTERFACE)
add_library(ZTO::MediaApi ALIAS zto_media_api)
set_target_properties(zto_media_api PROPERTIES EXPORT_NAME MediaApi)
target_link_libraries(zto_media_api INTERFACE ZTO::MediaTypes ZTO::Services)
target_include_directories(zto_media_api INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
set(package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/ZTOrbitMediaApi")
configure_package_config_file(ZTOrbitMediaApiConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ZTOrbitMediaApiConfig.cmake" INSTALL_DESTINATION "${package_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ZTOrbitMediaApiConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)
install(TARGETS zto_media_types EXPORT ZTOrbitMediaTypesTargets)
install(EXPORT ZTOrbitMediaTypesTargets NAMESPACE ZTO:: DESTINATION "${package_dir}")
install(TARGETS zto_media_api EXPORT ZTOrbitMediaApiTargets)
install(EXPORT ZTOrbitMediaApiTargets NAMESPACE ZTO:: DESTINATION "${package_dir}")
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(DIRECTORY ../examples/preview/ DESTINATION examples/preview)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ZTOrbitMediaApiConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/ZTOrbitMediaApiConfigVersion.cmake" DESTINATION "${package_dir}")
option(MEDIA_BUILD_QUICK "Build optional Qt Quick frame presenter" ON)
if(MEDIA_BUILD_QUICK)
    find_package(Qt6 REQUIRED COMPONENTS Quick)
    add_library(zto_media_quick STATIC quick/VideoPreview.cpp)
    add_library(ZTO::MediaQuick ALIAS zto_media_quick)
    set_target_properties(zto_media_quick PROPERTIES EXPORT_NAME MediaQuick)
    target_link_libraries(zto_media_quick PUBLIC ZTO::MediaTypes Qt6::Quick)
    install(TARGETS zto_media_quick EXPORT ZTOrbitMediaQuickTargets ARCHIVE DESTINATION lib)
    install(EXPORT ZTOrbitMediaQuickTargets NAMESPACE ZTO:: DESTINATION "${package_dir}")
endif()

cmake_minimum_required(VERSION 3.21.1)
project(ZTGUIMediaApi VERSION 2.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(ZTGUIPluginApi 0.5 EXACT CONFIG REQUIRED COMPONENTS Services)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)
add_library(ztgui_media_types INTERFACE)
add_library(ZTGUI::MediaTypes ALIAS ztgui_media_types)
set_target_properties(ztgui_media_types PROPERTIES EXPORT_NAME MediaTypes)
target_link_libraries(ztgui_media_types INTERFACE Qt6::Core)
target_compile_features(ztgui_media_types INTERFACE cxx_std_17)
target_include_directories(ztgui_media_types INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
add_library(ztgui_media_api INTERFACE)
add_library(ZTGUI::MediaApi ALIAS ztgui_media_api)
set_target_properties(ztgui_media_api PROPERTIES EXPORT_NAME MediaApi)
target_link_libraries(ztgui_media_api INTERFACE ZTGUI::MediaTypes ZTGUI::Services)
target_include_directories(ztgui_media_api INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
set(package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/ZTGUIMediaApi")
configure_package_config_file(ZTGUIMediaApiConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ZTGUIMediaApiConfig.cmake" INSTALL_DESTINATION "${package_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ZTGUIMediaApiConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)
install(TARGETS ztgui_media_types EXPORT ZTGUIMediaTypesTargets)
install(EXPORT ZTGUIMediaTypesTargets NAMESPACE ZTGUI:: DESTINATION "${package_dir}")
install(TARGETS ztgui_media_api EXPORT ZTGUIMediaApiTargets)
install(EXPORT ZTGUIMediaApiTargets NAMESPACE ZTGUI:: DESTINATION "${package_dir}")
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(DIRECTORY ../examples/preview/ DESTINATION examples/preview)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ZTGUIMediaApiConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/ZTGUIMediaApiConfigVersion.cmake" DESTINATION "${package_dir}")
option(MEDIA_BUILD_QUICK "Build optional Qt Quick frame presenter" ON)
if(MEDIA_BUILD_QUICK)
    find_package(Qt6 REQUIRED COMPONENTS Quick)
    add_library(ztgui_media_quick STATIC quick/VideoPreview.cpp)
    add_library(ZTGUI::MediaQuick ALIAS ztgui_media_quick)
    set_target_properties(ztgui_media_quick PROPERTIES EXPORT_NAME MediaQuick)
    target_link_libraries(ztgui_media_quick PUBLIC ZTGUI::MediaTypes Qt6::Quick)
    install(TARGETS ztgui_media_quick EXPORT ZTGUIMediaQuickTargets ARCHIVE DESTINATION lib)
    install(EXPORT ZTGUIMediaQuickTargets NAMESPACE ZTGUI:: DESTINATION "${package_dir}")
endif()

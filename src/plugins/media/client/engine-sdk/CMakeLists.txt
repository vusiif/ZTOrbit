cmake_minimum_required(VERSION 3.21.1)
project(ZTGUIMediaEngine VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(FFmpeg.cmake)
include(ExtensionBundle.cmake)
configure_file(Fingerprint.h.in "${CMAKE_CURRENT_BINARY_DIR}/include/ztmediaengine/Fingerprint.h" @ONLY)
add_library(ztmedia_engine INTERFACE)
add_library(ZTGUI::MediaEngine ALIAS ztmedia_engine)
set_target_properties(ztmedia_engine PROPERTIES EXPORT_NAME MediaEngine)
target_compile_features(ztmedia_engine INTERFACE cxx_std_17)
target_include_directories(ztmedia_engine INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>" "$<INSTALL_INTERFACE:include>")
target_link_libraries(ztmedia_engine INTERFACE FFmpeg::avutil)
set(package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/ZTGUIMediaEngine")
configure_package_config_file(ZTGUIMediaEngineConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ZTGUIMediaEngineConfig.cmake" INSTALL_DESTINATION "${package_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ZTGUIMediaEngineConfigVersion.cmake" VERSION 1.0.0 COMPATIBILITY ExactVersion)
install(TARGETS ztmedia_engine EXPORT ZTGUIMediaEngineTargets)
install(EXPORT ZTGUIMediaEngineTargets NAMESPACE ZTGUI:: DESTINATION "${package_dir}")
install(DIRECTORY include/ "${CMAKE_CURRENT_BINARY_DIR}/include/" DESTINATION include)
install(DIRECTORY "${MEDIA_FFMPEG_ROOT}/include/" DESTINATION ffmpeg/include)
install(DIRECTORY "${MEDIA_FFMPEG_ROOT}/lib/" DESTINATION ffmpeg/lib)
install(DIRECTORY "${MEDIA_FFMPEG_ROOT}/bin/" DESTINATION ffmpeg/bin)
install(FILES "${MEDIA_FFMPEG_ROOT}/actual-build.json" "${MEDIA_FFMPEG_ROOT}/sources.json" "${MEDIA_FFMPEG_ROOT}/COPYING.LGPLv2.1" "${MEDIA_FFMPEG_ROOT}/LICENSE.md" "${MEDIA_FFMPEG_ROOT}/LICENSE" DESTINATION ffmpeg)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ZTGUIMediaEngineConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/ZTGUIMediaEngineConfigVersion.cmake" FFmpeg.cmake ExtensionBundle.cmake DESTINATION "${package_dir}")
install(DIRECTORY ../examples/flip/ DESTINATION examples/flip)

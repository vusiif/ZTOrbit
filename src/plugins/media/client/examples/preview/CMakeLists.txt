cmake_minimum_required(VERSION 3.21.1)
project(ztmedia_preview_probe LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(ZTOrbitPluginApi 0.5 EXACT CONFIG REQUIRED)
find_package(ZTOrbitMediaApi 2 CONFIG REQUIRED COMPONENTS Quick)
qt_add_plugin(ztmedia_preview_probe SHARED CLASS_NAME PreviewPlugin Preview.cpp)
target_link_libraries(ztmedia_preview_probe PRIVATE ZTO::Api ZTO::MediaApi ZTO::MediaQuick)
configure_file(plugin.json.in "${CMAKE_CURRENT_BINARY_DIR}/plugin.json" @ONLY)
target_include_directories(ztmedia_preview_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
qt_add_resources(ztmedia_preview_probe preview PREFIX /ztmedia/probe FILES Preview.qml)
if(MSVC)
    target_compile_options(ztmedia_preview_probe PRIVATE /utf-8)
endif()
zto_package_plugin(ztmedia_preview_probe HOST client ID media-preview-probe NAME "媒体开发预览" VERSION "0.1.0"
    API "${ZTO_API_VERSION}" IID "vusiif.zto.api.PluginInterface/3.0" SERVICE_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/service-manifest.json")

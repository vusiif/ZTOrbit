cmake_minimum_required(VERSION 3.21.1)
project(ztmedia_flip LANGUAGES CXX)
if(NOT TARGET ZTO::MediaEngine)
    find_package(ZTOrbitMediaEngine 1.0.0 EXACT CONFIG REQUIRED)
endif()
add_library(ztmedia_flip SHARED Flip.cpp)
target_link_libraries(ztmedia_flip PRIVATE ZTO::MediaEngine FFmpeg::avfilter)
set_target_properties(ztmedia_flip PROPERTIES PREFIX "" OUTPUT_NAME "z2media_flip")
ztmedia_write_extension_bundle("${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/extension-bundle.json" ztmedia_flip)
install(TARGETS ztmedia_flip RUNTIME DESTINATION extensions/flip LIBRARY DESTINATION extensions/flip)

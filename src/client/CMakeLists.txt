cmake_minimum_required(VERSION 3.21.1)
include(../../cmake/Version.cmake)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(zto VERSION "${ZTO_VERSION_NUMERIC}" LANGUAGES CXX)
    set(ZTO_ROOT_DIR "${ZTO_REPOSITORY}")
    set(ZTO_BUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}")
    option(ZTO_BUILD_TESTS "Build client regression tests" OFF)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick Qml QuickControls2 Widgets Concurrent Network)
get_target_property(qt_type Qt6::Core TYPE)
if(qt_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "ZTOrbit requires shared Qt for runtime plugins")
endif()
add_subdirectory(api)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shared/network" "${CMAKE_CURRENT_BINARY_DIR}/shared-network")
add_subdirectory(main)
if(ZTO_BUILD_TESTS)
    enable_testing()
    add_subdirectory("${ZTO_ROOT_DIR}/src/tests" "${ZTO_BUILD_DIR}/tests")
endif()

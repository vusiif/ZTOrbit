cmake_minimum_required(VERSION 3.21.1)
if(TARGET ztgui_package)
    return()
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Network)
if(WIN32)
    set(ZTGUI_SODIUM_ROOT "" CACHE PATH "Unpacked official libsodium MSVC archive")
    if(NOT ZTGUI_SODIUM_ROOT)
        set(archive "${CMAKE_CURRENT_BINARY_DIR}/sodium.zip")
        file(DOWNLOAD https://download.libsodium.org/libsodium/releases/libsodium-1.0.20-msvc.zip "${archive}"
            EXPECTED_HASH SHA256=2ff97f9e3f5b341bdc808e698057bea1ae454f99e29ff6f9b62e14d0eb1b1baa TLS_VERIFY ON)
        file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/sodium")
        set(ZTGUI_SODIUM_ROOT "${CMAKE_CURRENT_BINARY_DIR}/sodium/libsodium")
    endif()
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(sodium_arch x64)
    else()
        set(sodium_arch Win32)
    endif()
    set(sodium_dir "${ZTGUI_SODIUM_ROOT}/${sodium_arch}/Release/v143/dynamic")
    add_library(ztgui_sodium SHARED IMPORTED GLOBAL)
    set_target_properties(ztgui_sodium PROPERTIES IMPORTED_LOCATION "${sodium_dir}/libsodium.dll"
        IMPORTED_IMPLIB "${sodium_dir}/libsodium.lib" INTERFACE_INCLUDE_DIRECTORIES "${ZTGUI_SODIUM_ROOT}/include")
else()
    find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
    find_library(SODIUM_LIBRARY NAMES sodium REQUIRED)
    add_library(ztgui_sodium UNKNOWN IMPORTED GLOBAL)
    set_target_properties(ztgui_sodium PROPERTIES IMPORTED_LOCATION "${SODIUM_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${SODIUM_INCLUDE_DIR}")
endif()
add_library(ztgui_package STATIC package/PluginPackage.cpp package/PluginBundle.cpp package/PluginRuntime.cpp package/PluginDependencies.cpp)
target_link_libraries(ztgui_package PUBLIC Qt6::Core PRIVATE ztgui_sodium)
target_include_directories(ztgui_package PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/package" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(ztgui_package PUBLIC cxx_std_17)
add_executable(ztplugin_pack package/PackageTool.cpp)
target_link_libraries(ztplugin_pack PRIVATE ztgui_package)
if(WIN32)
    add_custom_command(TARGET ztplugin_pack POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:ztgui_sodium>" "$<TARGET_FILE_DIR:ztplugin_pack>"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::Core>" "$<TARGET_FILE_DIR:ztplugin_pack>")
endif()
if(MSVC)
    target_compile_options(ztgui_package PRIVATE /utf-8)
endif()

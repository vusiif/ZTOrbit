cmake_minimum_required(VERSION 3.21.1)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ztserver_voice_chat LANGUAGES CXX)
    find_package(ZTServerPluginApi 0.3 EXACT CONFIG REQUIRED)
endif()
set(CMAKE_AUTOMOC ON)
set(voice_protocol "${CMAKE_CURRENT_SOURCE_DIR}/../../../shared" CACHE PATH "Shared voice protocol source directory")
if(NOT ZTSERVER_API_VERSION)
    set(ZTSERVER_API_VERSION "${ZTGUI_API_VERSION}")
endif()
configure_file(plugin.json.in "${CMAKE_CURRENT_BINARY_DIR}/plugin.json" @ONLY)
qt_add_plugin(ztserver_voice_chat SHARED CLASS_NAME vusiif::ztgui::voice_chat::VoiceServerPlugin)
target_sources(ztserver_voice_chat PRIVATE Plugin.cpp)
target_include_directories(ztserver_voice_chat PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${voice_protocol}")
target_link_libraries(ztserver_voice_chat PRIVATE ZTServer::Api)
set_target_properties(ztserver_voice_chat PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
install(TARGETS ztserver_voice_chat LIBRARY DESTINATION plugins)

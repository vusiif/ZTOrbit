cmake_minimum_required(VERSION 3.21.1)
include(../../cmake/Version.cmake)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(zt_server VERSION "${ZEROTIER_VERSION}" LANGUAGES CXX)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "zt-server supports Linux only; configure src/server/api independently for the SDK")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)
get_target_property(qt_type Qt6::Core TYPE)
if(qt_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "zt-server requires shared Qt")
endif()
add_subdirectory(api)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shared/network" "${CMAKE_CURRENT_BINARY_DIR}/shared-network")
add_subdirectory(main)
option(ZTSERVER_BUILD_VOICE_PLUGIN "Build the voice server plugin" OFF)
if(ZTSERVER_BUILD_VOICE_PLUGIN)
    add_subdirectory("${ZTGUI_REPOSITORY}/src/plugins/voice-chat/server" "${CMAKE_CURRENT_BINARY_DIR}/voice-chat")
endif()
option(ZTSERVER_BUILD_TESTS "Build isolated server tests and fixture plugins" OFF)
if(ZTSERVER_BUILD_TESTS)
    enable_testing()
    add_subdirectory("${ZTGUI_REPOSITORY}/src/tests/server" "${CMAKE_BINARY_DIR}/tests")
endif()

find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
add_executable(server_package_tests PackageTests.cpp)
target_link_libraries(server_package_tests PRIVATE ztgui_package Qt6::Test ZTServer::Api)
target_compile_definitions(server_package_tests PRIVATE FIXTURE_LIBRARY="$<TARGET_FILE:server_fixture_echo>")
add_dependencies(server_package_tests server_fixture_echo)
add_test(NAME server_package_container COMMAND server_package_tests)
add_executable(server_query_tests QueryTests.cpp "${CMAKE_CURRENT_SOURCE_DIR}/../main/ZeroTierQueries.cpp")
target_include_directories(server_query_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../main")
target_link_libraries(server_query_tests PRIVATE ZTServer::Api Qt6::Test)
add_test(NAME server_queries COMMAND server_query_tests)
foreach(kind echo other failed bad duplicate cancellable)
    set(FIXTURE_API "${ZTGUI_API_VERSION}")
    if(kind STREQUAL "bad")
        set(FIXTURE_API "0.999")
    endif()
    configure_file(fixture.json.in "${CMAKE_CURRENT_BINARY_DIR}/${kind}/fixture.json" @ONLY)
    qt_add_plugin(server_fixture_${kind} SHARED CLASS_NAME Fixture)
    target_sources(server_fixture_${kind} PRIVATE Fixture.cpp)
    target_link_libraries(server_fixture_${kind} PRIVATE ZTServer::Api)
    target_include_directories(server_fixture_${kind} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${kind}")
    set(id "test-${kind}")
    if(kind STREQUAL "duplicate")
        set(id "test-echo")
    endif()
    target_compile_definitions(server_fixture_${kind} PRIVATE FIXTURE_ID="${id}")
    if(kind STREQUAL "failed")
        target_compile_definitions(server_fixture_${kind} PRIVATE FIXTURE_FAILED=1)
    endif()
    if(kind STREQUAL "cancellable")
        target_compile_definitions(server_fixture_${kind} PRIVATE FIXTURE_CANCELLABLE=1)
    endif()
    set_target_properties(server_fixture_${kind} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixtures")
    add_dependencies(server_fixture_${kind} ztplugin_pack)
    add_custom_command(TARGET server_fixture_${kind} POST_BUILD COMMAND "$<TARGET_FILE:ztplugin_pack>"
        --library "$<TARGET_FILE:server_fixture_${kind}>" --output "$<TARGET_FILE_DIR:server_fixture_${kind}>/server_fixture_${kind}.ztplugin"
        --host server --id "${id}" --name "${id}" --version 1 --apiVersion "${FIXTURE_API}" --iid "vusiif.ztgui.server.api.PluginInterface/2.0" VERBATIM)
endforeach()

cmake_minimum_required(VERSION 3.21.1)
include("${CMAKE_CURRENT_LIST_DIR}/Version.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ztohub VERSION "${ZTO_VERSION_NUMERIC}" LANGUAGES CXX)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "ztohub supports Linux only; configure src/server-old/api independently for the SDK")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)
get_target_property(qt_type Qt6::Core TYPE)
if(qt_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "ztohub requires shared Qt")
endif()
add_subdirectory(api)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shared/network" "${CMAKE_CURRENT_BINARY_DIR}/shared-network")
add_subdirectory(main)
option(ZTOHUB_BUILD_VOICE_PLUGIN "Build the voice server plugin" OFF)
if(ZTOHUB_BUILD_VOICE_PLUGIN)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/plugins/voice-chat/server" "${CMAKE_CURRENT_BINARY_DIR}/voice-chat")
endif()
option(ZTOHUB_BUILD_TESTS "Build isolated server tests and fixture plugins" OFF)
if(ZTOHUB_BUILD_TESTS)
    enable_testing()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/tests" "${CMAKE_BINARY_DIR}/tests")
endif()

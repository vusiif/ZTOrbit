cmake_minimum_required(VERSION 3.21.1)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ztohub_voice_chat LANGUAGES CXX)
    find_package(ZTOrbitHubPluginApi 0.4 EXACT CONFIG REQUIRED)
endif()
set(CMAKE_AUTOMOC ON)
set(voice_protocol "${CMAKE_CURRENT_SOURCE_DIR}/../../../../plugins/voice-chat/shared" CACHE PATH "Shared voice protocol source directory")
if(NOT ZTOHUB_API_VERSION)
    set(ZTOHUB_API_VERSION "${ZTO_API_VERSION}")
endif()
configure_file(plugin.json.in "${CMAKE_CURRENT_BINARY_DIR}/plugin.json" @ONLY)
qt_add_plugin(ztohub_voice_chat SHARED CLASS_NAME vusiif::zto::voice_chat::VoiceServerPlugin)
target_sources(ztohub_voice_chat PRIVATE Plugin.cpp)
target_include_directories(ztohub_voice_chat PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${voice_protocol}")
target_link_libraries(ztohub_voice_chat PRIVATE ZTOrbitHub::Api)
set_target_properties(ztohub_voice_chat PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")

zto_package_plugin(ztohub_voice_chat HOST server ID voice-chat NAME "语音聊天" VERSION "0.1.0" API "${ZTOHUB_API_VERSION}" IID "vusiif.ztohub.api.PluginInterface/2.0")
install(FILES "$<TARGET_FILE_DIR:ztohub_voice_chat>/ztohub_voice_chat.ztplugin" DESTINATION plugins)

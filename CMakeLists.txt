cmake_minimum_required(VERSION 3.21.1)
include(cmake/Version.cmake)
project(zerotier_gui VERSION "${ZEROTIER_VERSION}" LANGUAGES CXX)
set(ZTGUI_ROOT_DIR "${PROJECT_SOURCE_DIR}")
set(ZTGUI_BUILD_DIR "${PROJECT_BINARY_DIR}")

option(ZTGUI_BUILD_CLIENT "Build the desktop client" ON)
option(ZTGUI_BUILD_SERVER "Build the Linux server" OFF)
option(ZEROTIER_BUILD_TESTS "Build client regression tests" OFF)
if(ZEROTIER_BUILD_TESTS)
    enable_testing()
endif()
if(ZTGUI_BUILD_CLIENT)
    add_subdirectory(src/client)
endif()
if(ZTGUI_BUILD_SERVER)
    add_subdirectory(src/server)
endif()

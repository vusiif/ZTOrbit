cmake_minimum_required(VERSION 3.21.1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ZEROTIER_VERSION)
string(STRIP "${ZEROTIER_VERSION}" ZEROTIER_VERSION)
if(NOT ZEROTIER_VERSION MATCHES "^0\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "Pre-release VERSION must be 0.core.patch")
endif()
project(zerotier_gui VERSION "${ZEROTIER_VERSION}" LANGUAGES CXX)
set(ZTGUI_ROOT_DIR "${PROJECT_SOURCE_DIR}")
set(ZTGUI_BUILD_DIR "${PROJECT_BINARY_DIR}")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Concurrent)
get_target_property(QT_CORE_LIBRARY_TYPE Qt6::Core TYPE)
if(QT_CORE_LIBRARY_TYPE STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "ZTGUI requires shared Qt for runtime plugins")
endif()

add_subdirectory(src/api)
add_subdirectory(src/main)
option(ZEROTIER_BUILD_TESTS "Build GUI regression tests with a fake ZeroTier CLI" OFF)
if(ZEROTIER_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

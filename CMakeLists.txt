cmake_minimum_required(VERSION 3.21.1)
include(cmake/Version.cmake)
project(zto VERSION "${ZTO_VERSION_NUMERIC}" LANGUAGES CXX)
set(ZTO_ROOT_DIR "${PROJECT_SOURCE_DIR}")
set(ZTO_BUILD_DIR "${PROJECT_BINARY_DIR}")

option(ZTO_BUILD_CLIENT "Build the desktop client" ON)
option(ZTO_BUILD_SERVER_OLD "Build archived C++ server" OFF)
option(ZTO_BUILD_TESTS "Build client regression tests" OFF)
if(ZTO_BUILD_TESTS)
    enable_testing()
endif()
if(ZTO_BUILD_CLIENT)
    add_subdirectory(src/client)
endif()
if(ZTO_BUILD_SERVER_OLD)
    add_subdirectory(src/server-old)
endif()
